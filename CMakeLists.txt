cmake_minimum_required(VERSION 3.22)
project(ReverieEnsemble VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(REVERIE_BUILD_GUI "Build the Qt Widgets desktop application" ON)
option(REVERIE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(REVERIE_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(miniaudio STATIC src/platform/miniaudio.c)
target_include_directories(miniaudio PUBLIC third_party/miniaudio)
if(APPLE)
  target_link_libraries(miniaudio PUBLIC "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreFoundation")
elseif(UNIX)
  target_link_libraries(miniaudio PUBLIC dl m)
endif()
add_library(reverie_core src/core/Export.cpp src/core/MidiFile.cpp src/core/Hash.cpp src/core/Library.cpp src/core/Effects.cpp src/core/Model.cpp src/core/Dsp.cpp src/core/Engine.cpp src/core/Media.cpp src/core/Serializer.cpp src/core/Recorder.cpp)
target_include_directories(reverie_core PUBLIC src PRIVATE third_party)
target_link_libraries(reverie_core PUBLIC miniaudio Threads::Threads)
target_compile_options(reverie_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(reverie-render src/render_main.cpp)
target_link_libraries(reverie-render PRIVATE reverie_core)
add_executable(reverie-benchmark tests/audio_benchmark.cpp)
target_link_libraries(reverie-benchmark PRIVATE reverie_core)
add_executable(reverie-calibrate src/calibrate_main.cpp)
target_include_directories(reverie-calibrate PRIVATE third_party)
target_link_libraries(reverie-calibrate PRIVATE reverie_core)
add_executable(reverie-demo03 src/demo03_main.cpp)
target_link_libraries(reverie-demo03 PRIVATE reverie_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(reverie-features03-tests tests/features03_tests.cpp)
  target_link_libraries(reverie-features03-tests PRIVATE reverie_core)
  target_compile_definitions(reverie-features03-tests PRIVATE REVERIE_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME features-03 COMMAND reverie-features03-tests)
  add_executable(reverie-core-tests tests/core_tests.cpp)
  target_link_libraries(reverie-core-tests PRIVATE reverie_core)
  add_test(NAME audio-core COMMAND reverie-core-tests)
  add_executable(reverie-features02-tests tests/features02_tests.cpp)
  target_link_libraries(reverie-features02-tests PRIVATE reverie_core)
  target_compile_definitions(reverie-features02-tests PRIVATE REVERIE_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME features-02 COMMAND reverie-features02-tests)
endif()
if(REVERIE_BUILD_GUI)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)
  if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  endif()
  add_library(reverie_ui src/ui/InstrumentEditor.cpp src/ui/InstrumentArt.cpp src/ui/ExportDialog.cpp src/ui/SettingsDialog.cpp src/ui/AudioEditor.cpp src/ui/InspectorPanel.cpp src/ui/Theme.cpp src/ui/Workspace.cpp src/ui/SoundBrowser.cpp src/ui/MixerView.cpp src/ui/Editor.cpp src/ui/Timeline.cpp src/ui/PianoRoll.cpp src/platform/AudioDevice.cpp)
  set_target_properties(reverie_ui PROPERTIES AUTOMOC ON)
  target_compile_definitions(reverie_ui PRIVATE REVERIE_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  target_link_libraries(reverie_ui PUBLIC reverie_core Qt6::Widgets)
  target_compile_options(reverie_ui PRIVATE -Wall -Wextra -Wpedantic)
  if(APPLE)
    enable_language(OBJCXX)
    target_sources(reverie_ui PRIVATE src/platform/Microphone.mm)
    target_link_libraries(reverie_ui PRIVATE "-framework AVFoundation")
  endif()
  add_executable(ReverieEnsemble MACOSX_BUNDLE src/main.cpp)
  target_link_libraries(ReverieEnsemble PRIVATE reverie_ui)
  set_target_properties(ReverieEnsemble PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Reverie Ensemble" MACOSX_BUNDLE_GUI_IDENTIFIER "com.quadraj.reverie-ensemble" MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION} MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION} MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/platform/Info.plist.in")
  if(APPLE)
    add_custom_command(TARGET ReverieEnsemble POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:ReverieEnsemble>/Contents/Resources/Legal"
      COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE" "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_LICENSES.md" "$<TARGET_BUNDLE_DIR:ReverieEnsemble>/Contents/Resources/Legal"
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/third_party/licenses" "$<TARGET_BUNDLE_DIR:ReverieEnsemble>/Contents/Resources/Legal/licenses"
      COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/third_party/miniaudio/LICENSE" "$<TARGET_BUNDLE_DIR:ReverieEnsemble>/Contents/Resources/Legal/licenses/miniaudio.txt")
  endif()
  if(APPLE)
    add_custom_command(TARGET ReverieEnsemble POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_BUNDLE_DIR:ReverieEnsemble>/Contents/Resources/assets")
  endif()
  if(APPLE)
    file(GLOB_RECURSE REVERIE_ASSET_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/*")
    set_property(TARGET ReverieEnsemble APPEND PROPERTY LINK_DEPENDS ${REVERIE_ASSET_FILES} "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE" "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_LICENSES.md")
  endif()
  if(BUILD_TESTING)
    add_executable(reverie-ui03-tests tests/ui03_tests.cpp)
    target_link_libraries(reverie-ui03-tests PRIVATE reverie_ui Qt6::Test)
    target_compile_definitions(reverie-ui03-tests PRIVATE REVERIE_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME editor-03 COMMAND reverie-ui03-tests)
    set_tests_properties(editor-03 PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(reverie-ui-tests tests/ui_tests.cpp)
    target_link_libraries(reverie-ui-tests PRIVATE reverie_ui Qt6::Test)
    add_test(NAME editor-workflow COMMAND reverie-ui-tests)
    add_executable(reverie-ui02-tests tests/ui02_tests.cpp)
    target_link_libraries(reverie-ui02-tests PRIVATE reverie_ui Qt6::Test)
    target_compile_definitions(reverie-ui02-tests PRIVATE REVERIE_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME editor-02 COMMAND reverie-ui02-tests)
    set_tests_properties(editor-02 PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    set_tests_properties(editor-workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  install(TARGETS ReverieEnsemble BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
install(TARGETS reverie-render RUNTIME DESTINATION bin)
install(FILES LICENSE NOTICE THIRD_PARTY_LICENSES.md DESTINATION share/reverie-ensemble)

install(DIRECTORY third_party/licenses DESTINATION share/reverie-ensemble)
install(FILES third_party/miniaudio/LICENSE DESTINATION share/reverie-ensemble/licenses RENAME miniaudio.txt)
